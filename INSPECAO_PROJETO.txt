from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session

from app.infrastructure.database.database import SessionLocal
from app.domain.models.usuario import Usuario
from app.core.token import verificar_token


oauth2_scheme = HTTPBearer()

def get_db():
    """Fornece uma sessÃ£o do banco de dados."""

    db = SessionLocal()

    try:
        yield db
    finally:
        db.close()


def get_usuario_atual(
    credenciais: HTTPAuthorizationCredentials = Depends(oauth2_scheme),
    db: Session = Depends(get_db)
):
    """Identifica o usuÃ¡rio por meio do token JWT."""

    credenciais_invalidas = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Token invÃ¡lido ou expirado.",
        headers={"WWW-Authenticate": "Bearer"}
    )
    token = credenciais.credentials
    dados_token = verificar_token(token)

    if dados_token is None:
        raise credenciais_invalidas

    usuario_id = dados_token.get("sub")

    if usuario_id is None:
        raise credenciais_invalidas

    try:
        usuario_id = int(usuario_id)
    except (ValueError, TypeError):
        raise credenciais_invalidas

    usuario = db.get(Usuario, usuario_id)

    if usuario is None or not usuario.ativo:
        raise credenciais_invalidas

    return usuario


def exigir_perfil(*perfis_permitidos: str):
    """
    Verifica se o usuÃ¡rio autenticado possui
    um dos perfis autorizados.
    """

    def verificar_permissao(
        usuario: Usuario = Depends(get_usuario_atual)
    ):
        if usuario.perfil not in perfis_permitidos:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="VocÃª nÃ£o possui permissÃ£o para realizar esta operaÃ§Ã£o."
            )

        return usuario

    return verificar_permissao

    UsuarioCriacao,

from app.core.token import criar_token_acesso


router = APIRouter(
    prefix="/auth",
    tags=["AutenticaÃ§Ã£o"]
)


@router.post(
    "/cadastro",
    response_model=UsuarioResposta,
    status_code=status.HTTP_201_CREATED
)
def cadastrar_usuario(
    dados: UsuarioCriacao,
    db: Session = Depends(get_db)
):
    email_existente = db.query(Usuario).filter(
        Usuario.email == dados.email
    ).first()

    if email_existente:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Este email jÃ¡ estÃ¡ cadastrado."
        )

    usuario = Usuario(
        nome=dados.nome,
        email=dados.email,
        senha_hash=gerar_hash_senha(dados.senha),
        perfil="CLIENTE"
    )

    db.add(usuario)
    db.commit()
    db.refresh(usuario)

    return usuario


@router.post("/login")
def realizar_login(
    dados: UsuarioLogin,
    db: Session = Depends(get_db)
):
    usuario = db.query(Usuario).filter(
        Usuario.email == dados.email
    ).first()

    if not usuario or not verificar_senha(
        dados.senha,
        usuario.senha_hash
    ):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Email ou senha incorretos."
        )

    if not usuario.ativo:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="UsuÃ¡rio inativo."
        )

    token = criar_token_acesso({
        "sub": str(usuario.id),
        "perfil": usuario.perfil
    })

    return {
        "access_token": token,
        "token_type": "bearer"
    }


@router.get("/me", response_model=UsuarioResposta)
def obter_usuario_logado(
    usuario: Usuario = Depends(get_usuario_atual)
):
    return usuario

@router.post(
    "/{pedido_id}/simular",
    response_model=PagamentoResposta,
    status_code=201
)
def simular_pagamento_endpoint(
    pedido_id: int,
    dados: PagamentoSimulacao,
    db: Session = Depends(get_db),
    usuario_atual: Usuario = Depends(get_usuario_atual)
):

    try:
        # 1. Identificar o perfil do usuÃ¡rio
        perfil = usuario_atual.perfil

        if hasattr(perfil, "value"):
            perfil = perfil.value

        # 2. Buscar o pedido
        pedido = buscar_pedido(db, pedido_id)

        # 3. Verificar autorizaÃ§Ã£o
        if perfil == "CLIENTE":

            if pedido.cliente_id != usuario_atual.id:
                raise HTTPException(
                    status_code=403,
                    detail="VocÃª nÃ£o pode pagar este pedido."
                )

        elif perfil == "ATENDENTE":

            if pedido.atendente_id != usuario_atual.id:
                raise HTTPException(
                    status_code=403,
                    detail="VocÃª nÃ£o pode processar o pagamento deste pedido."
                )

        else:
            raise HTTPException(
                status_code=403,
                detail="Perfil nÃ£o autorizado a processar pagamentos."
            )

        # 4. Processar o pagamento
        pagamento = processar_pagamento(
            db,
            pedido_id,
            dados.aprovado
        )

        # 5. Confirmar transaÃ§Ã£o
        db.commit()
        db.refresh(pagamento)

        return pagamento

    except HTTPException:
        db.rollback()
        raise

    except ValueError as erro:
        db.rollback()
        raise HTTPException(
            status_code=400,
            detail=str(erro)
        )

    except Exception:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Erro interno ao processar pagamento."
        )

from decimal import Decimal, ROUND_DOWN
from sqlalchemy.orm import Session

from app.domain.models.consentimento import Consentimento
from app.domain.models.ponto_fidelidade import PontoFidelidade


FINALIDADE_FIDELIDADE = "PROGRAMA_FIDELIDADE"


def cliente_aderiu_fidelidade(
    db: Session,
    cliente_id: int
) -> bool:

    consentimento = db.query(Consentimento).filter(
        Consentimento.usuario_id == cliente_id,
        Consentimento.finalidade == FINALIDADE_FIDELIDADE,
        Consentimento.autorizado == True,
        Consentimento.data_revogacao.is_(None)
    ).first()

    return consentimento is not None


def calcular_pontos(valor: Decimal) -> int:
    """
    Regra MVP:
    1 ponto por R$ 1,00 pago.
    FraÃ§Ãµes de reais sÃ£o arredondadas para baixo.
    """

    return int(
        valor.quantize(
            Decimal("1"),
            rounding=ROUND_DOWN
        )
    )


def creditar_pontos(
    db: Session,
    pedido,
    pagamento
):
    # 1. Somente pagamentos aprovados geram pontos.
    if pagamento.status_pagamento.value != "APROVADO":
        return None

    # 2. Verificar se o cliente aderiu ao programa.
    if not cliente_aderiu_fidelidade(
        db,
        pedido.cliente_id
    ):
        return None

    # 3. Prevenir crÃ©dito duplicado.
    credito_existente = db.query(PontoFidelidade).filter(
        PontoFidelidade.pedido_id == pedido.id
    ).first()

    if credito_existente:
        return credito_existente

    # 4. Calcular pontos com base no pagamento.
    pontos = calcular_pontos(pagamento.valor)

    if pontos <= 0:
        return None

    # 5. Registrar crÃ©dito no histÃ³rico.
    credito = PontoFidelidade(
        cliente_id=pedido.cliente_id,
        pedido_id=pedido.id,
        pagamento_id=pagamento.id,
        pontos=pontos
    )

    db.add(credito)

    # NÃ£o faz commit aqui.
    # O commit serÃ¡ realizado pelo endpoint de pagamento.

    return credito

from sqlalchemy.orm import Session
from sqlalchemy import func

from app.domain.models.pedido import Pedido

from app.domain.enums.status_pedido import StatusPedido
from app.domain.enums.status_pagamento import StatusPagamento

from app.application.services.fidelidade_service import creditar_pontos


def buscar_pedido(db: Session, pedido_id: int) -> Pedido:
    pedido = db.query(Pedido).filter(
        Pedido.id == pedido_id
    ).first()

    if not pedido:
        raise ValueError("Pedido nÃ£o encontrado.")

    return pedido


def validar_pedido_para_pagamento(pedido: Pedido):
    if pedido.status != StatusPedido.AGUARDANDO_PAGAMENTO:
        raise ValueError(
            "Pedido nÃ£o estÃ¡ aguardando pagamento."
        )

    if not pedido.itens:
        raise ValueError("Pedido nÃ£o possui itens.")

    if pedido.valor_total <= 0:
        raise ValueError("Valor do pedido invÃ¡lido.")


def obter_numero_tentativa(
    db: Session,
    pedido_id: int
) -> int:
    ultima_tentativa = db.query(
        func.max(Pagamento.numero_tentativa)
    ).filter(
        Pagamento.pedido_id == pedido_id
    ).scalar()

    return (ultima_tentativa or 0) + 1


def validar_estoque_pedido(
    db: Session,
    pedido: Pedido
):
    for item in pedido.itens:
        estoque = db.query(Estoque).filter(
            Estoque.produto_unidade_id
            == item.produto_unidade_id
        ).first()

        if not estoque:
            raise ValueError(
                "Estoque nÃ£o encontrado para um item do pedido."
            )

        if estoque.quantidade_disponivel < item.quantidade:
            raise ValueError(
                "Estoque insuficiente para pagamento."
            )


def baixar_estoque_pedido(
    db: Session,
    pedido: Pedido
):
    for item in pedido.itens:
        estoque = db.query(Estoque).filter(
            Estoque.produto_unidade_id
            == item.produto_unidade_id
        ).first()

        if not estoque:
            raise ValueError(
                "Estoque nÃ£o encontrado para um item do pedido."
            )

        estoque.quantidade_disponivel -= item.quantidade


def processar_pagamento(
    db: Session,
    pedido_id: int,
    aprovado: bool
) -> Pagamento:

    # 1. Buscar o pedido
    pedido = buscar_pedido(db, pedido_id)

    # 2. Validar se o pedido pode receber pagamento
    validar_pedido_para_pagamento(pedido)

    # 3. Identificar o nÃºmero da tentativa
    tentativa = obter_numero_tentativa(
        db,
        pedido_id
    )

    # 4. Definir o resultado da simulaÃ§Ã£o
    if aprovado:

        # Verificar estoque antes da aprovaÃ§Ã£o
        validar_estoque_pedido(db, pedido)

        status_pagamento = StatusPagamento.APROVADO

        # Baixar estoque apÃ³s aprovaÃ§Ã£o simulada
        baixar_estoque_pedido(db, pedido)

        # Atualizar o status do pedido
        pedido.status = StatusPedido.EM_PREPARO

    else:
        # Pagamento recusado: nÃ£o baixa estoque
        status_pagamento = StatusPagamento.RECUSADO

    # 5. Registrar a tentativa de pagamento
    pagamento = Pagamento(
        pedido_id=pedido.id,
        status_pagamento=status_pagamento,
        valor=pedido.valor_total,
        numero_tentativa=tentativa,
        referencia_transacao=(
            f"SIM-{pedido.id}-{tentativa}"
        )
    )

    db.add(pagamento)

    # 6. Persistir dentro da transaÃ§Ã£o atual
    # O flush disponibiliza o ID do pagamento
    db.flush()

    # 7. Creditar pontos somente se aprovado
    # O serviÃ§o verifica o consentimento do cliente
    if aprovado:
        creditar_pontos(
            db=db,
            pedido=pedido,
            pagamento=pagamento
        )

    # O commit serÃ¡ realizado pela rota de pagamentos
    return pagamento

def validar_produto_unidade(
    db: Session,
    unidade_id: int,
    produto_unidade_id: int
) -> tuple[ProdutoUnidade, Produto]:

    associacao = db.query(ProdutoUnidade).filter(
        ProdutoUnidade.id == produto_unidade_id,
        ProdutoUnidade.unidade_id == unidade_id
    ).first()

    if associacao is None:
        raise ValueError(
            "Produto nÃ£o encontrado ou nÃ£o associado a esta unidade."
        )

    if not associacao.habilitado:
        raise ValueError(
            "Este produto estÃ¡ desabilitado nesta unidade."
        )

    hoje = date.today()

    if (
        associacao.data_inicio_disponibilidade is not None
        and hoje < associacao.data_inicio_disponibilidade
    ):
        raise ValueError(
            "Este produto ainda nÃ£o estÃ¡ disponÃ­vel."
        )

    if (
        associacao.data_fim_disponibilidade is not None
        and hoje > associacao.data_fim_disponibilidade
    ):
        raise ValueError(
            "O perÃ­odo de disponibilidade deste produto terminou."
        )

    produto = db.get(Produto, associacao.produto_id)

    if produto is None or not produto.ativo:
        raise ValueError(
            "Este produto estÃ¡ inativo ou nÃ£o foi encontrado."
        )

    return associacao, produto

def calcular_preco_item(
    associacao: ProdutoUnidade,
    produto: Produto,
    quantidade: int
) -> tuple[Decimal, Decimal]:

    if associacao.preco_especifico is not None:
        preco_unitario = associacao.preco_especifico
    else:
        preco_unitario = produto.preco_padrao

    subtotal = preco_unitario * quantidade

    return preco_unitario, subtotal


def preparar_itens_pedido(
    db: Session,
    unidade_id: int,
    itens: list[ItemPedidoCriacao]
) -> tuple[list[tuple[int, int, Decimal, Decimal]], Decimal]:

    produtos_processados = set()
    itens_preparados = []
    valor_total = Decimal("0.00")

    for item in itens:

        if item.produto_unidade_id in produtos_processados:
            raise ValueError(
                "O mesmo produto nÃ£o pode ser repetido no pedido."
            )

        produtos_processados.add(item.produto_unidade_id)

        associacao, produto = validar_produto_unidade(
            db,
            unidade_id,
            item.produto_unidade_id
        )

        validar_estoque(
            db,
            item.produto_unidade_id,
            item.quantidade
        )

        preco_unitario, subtotal = calcular_preco_item(
            associacao,
            produto,
            item.quantidade
        )

        itens_preparados.append(
            (
                associacao.id,
                item.quantidade,
                preco_unitario,
                subtotal
            )
        )

        valor_total += subtotal

    return itens_preparados, valor_total

        validar_cliente(db, cliente_id)
        validar_atendente(db, usuario_atual.id)

        return cliente_id, usuario_atual.id

    raise ValueError("Perfil nÃ£o autorizado a criar pedidos.")

import jwt

def criar_token_acesso(dados: dict) -> str:
    """Gera um token JWT com prazo de validade."""

    dados_token = dados.copy()

    expiracao = datetime.now(timezone.utc) + timedelta(

    )

    token = jwt.encode(
        dados_token,
        settings.SECRET_KEY,
        algorithm=settings.ALGORITHM
    )

    return token


def verificar_token(token: str) -> dict | None:
    """Valida o token e retorna seus dados."""

    try:
        dados = jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM]
        )

        return dados

    except jwt.InvalidTokenError:
        return None
